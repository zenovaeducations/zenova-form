async function findStudent() {

    const entered =
        znvCode.value.trim();

    codeError.classList.add(
        "hidden"
    );

    /*
       Minimum 4 digits
    */

    if (
        !/^\d{4,}$/.test(entered)
    ) {

        showError(
            codeError,
            "Please enter at least 4 digits. Example: 2701"
        );

        return;

    }

    findStudentBtn.disabled = true;

    findStudentBtn.textContent =
        "Checking...";

    try {

        /*
          We read submissions once and
          compare the numeric portion locally.

          This supports:
          ZNV/SSLCM+2701
          ZNV/SSLCM+2798
          ZNV/SSLCM+27964
        */

        const snapshot =
            await getDocs(
                collection(
                    db,
                    "submissions"
                )
            );

        let found = null;

        snapshot.forEach(
            docSnap => {

                if (found) return;

                const data =
                    docSnap.data();

                const storedCode =
                    data.studentCode ||
                    data.student?.studentCode ||
                    data.code ||
                    "";

                const storedNumber =
                    extractCodeNumber(
                        storedCode
                    );

                if (
                    storedNumber ===
                    entered
                ) {

                    found = {

                        id:
                            docSnap.id,

                        ...data

                    };

                }

            }
        );

        if (!found) {

            showError(
                codeError,
                "We could not find a student with this ZNV number. Please check the number and try again."
            );

            return;

        }

        matchedStudent =
            found;

        const name =
            getStudentName(found);

        if (!name) {

            showError(
                codeError,
                "This student record does not have a registered name. Please contact Zenova."
            );

            return;

        }

        document.getElementById(
            "matchedName"
        ).textContent = name;

        document.getElementById(
            "matchedCode"
        ).textContent =
            "ZNV/SSLCM+" +
            entered;

        showScreen(
            "name",
            2
        );

    } catch (error) {

        console.error(error);

        showError(
            codeError,
            "Unable to check your ZNV number. Please try again."
        );

    } finally {

        findStudentBtn.disabled = false;

        findStudentBtn.textContent =
            "Continue";

    }

}

function calculateResult() {

    const errorBox =
        document.getElementById(
            "marksError"
        );

    errorBox.classList.add(
        "hidden"
    );

    const marks =
        getMarks();

    /*
      Validate exact maximums.
    */

    if (
        marks.kannada < 0 ||
        marks.kannada > 100
    ) {

        showError(
            errorBox,
            "Kannada marks must be between 0 and 100."
        );

        return;

    }

    const otherSubjects = [

        [
            "English",
            marks.english
        ],

        [
            "Hindi",
            marks.hindi
        ],

        [
            "Mathematics",
            marks.mathematics
        ],

        [
            "Science",
            marks.science
        ],

        [
            "Social Science",
            marks.socialScience
        ]

    ];

    for (
        const [
            subject,
            value
        ] of otherSubjects
    ) {

        if (
            value < 0 ||
            value > 80
        ) {

            showError(
                errorBox,
                subject +
                " marks must be between 0 and 80."
            );

            return;

        }

    }

    /*
      Make sure every field was entered.
    */

    const rawInputs =
        Array.from(
            document.querySelectorAll(
                ".mark"
            )
        );

    const missing =
        rawInputs.some(
            input =>
                input.value === ""
        );

    if (missing) {

        showError(
            errorBox,
            "Please enter marks for all 6 subjects."
        );

        return;

    }

    currentTotal =
        marks.kannada +
        marks.english +
        marks.hindi +
        marks.mathematics +
        marks.science +
        marks.socialScience;

    /*
      Total maximum = 500

      100 + 80 + 80 + 80 + 80 + 80
    */

    currentPercentage =
        (currentTotal / 500) * 100;

    currentPercentage =
        round(
            currentPercentage,
            2
        );

    /*
      Your requested conservative estimate:
      Midterm percentage - 10 percentage points.

      Example:
      70% → 60%
    */

    let estimated =
        currentPercentage - 10;

    if (estimated < 0) {
        estimated = 0;
    }

    estimated =
        round(
            estimated,
            2
        );

    document.getElementById(
        "percentage"
    ).textContent =
        currentPercentage + "%";

    document.getElementById(
        "resultTotal"
    ).textContent =
        currentTotal +
        " / 500";

    document.getElementById(
        "resultPercentage"
    ).textContent =
        currentPercentage +
        "%";

    document.getElementById(
        "estimatedPercentage"
    ).textContent =
        estimated +
        "%";

    createMotivation(
        currentPercentage
    );

    showScreen(
        "result",
        5
    );

}

function extractCodeNumber(
    code
) {

    if (!code)
        return "";

    /*
      ZNV/SSLCM+2701
      ZNV/SSLCM+27964
      2701

      All become:
      2701 / 27964
    */

    const text =
        String(code);

    const match =
        text.match(
            /(\d+)$/
        );

    return match
        ? match[1]
        : "";

}

function getStudentName(
    student
) {

    return firstValue([

        student.name,

        student.studentName,

        student.fullName,

        student.student?.name,

        student.student?.studentName,

        student.student?.fullName

    ]);

}

function getStudentPhone(
    student
) {

    return firstValue([

        student.phone,

        student.mobile,

        student.mobileNumber,

        student.phoneNumber,

        student.student?.phone,

        student.student?.mobile,

        student.student?.mobileNumber,

        student.student?.phoneNumber

    ]);

}

function normalizePhone(
    value
) {

    if (!value)
        return "";

    let digits =
        String(value)
            .replace(
                /\D/g,
                ""
            );

    /*
      +91XXXXXXXXXX
    */

    if (
        digits.length === 12 &&
        digits.startsWith("91")
    ) {

        digits =
            digits.substring(2);

    }

    return digits;

}

function firstValue(
    values
) {

    for (
        const value of values
    ) {

        if (
            value !== undefined &&
            value !== null &&
            String(value).trim() !== ""
        ) {

            return value;

        }

    }

    return "";

}

function round(
    number,
    decimals
) {

    const factor =
        Math.pow(
            10,
            decimals
        );

    return Math.round(
        number * factor
    ) / factor;

}

function showError(
    element,
    message
) {

    element.textContent =
        message;

    element.classList.remove(
        "hidden"
    );

}
